add_library(${PROJECT_NAME}-rfattenuator-interface INTERFACE)

set(RFATTENUATOR_INTERFACE_PUBLIC_INCLUDE ${CMAKE_CURRENT_LIST_DIR}/include)
set(RFATTENUATOR_INTERFACE_PUBLIC_INCLUDE_SUFFIX microsoft/net/remote/service)
set(RFATTENUATOR_INTERFACE_PUBLIC_INCLUDE_PREFIX ${RFATTENUATOR_INTERFACE_PUBLIC_INCLUDE}/${RFATTENUATOR_INTERFACE_PUBLIC_INCLUDE_SUFFIX})

target_sources(${PROJECT_NAME}-rfattenuator-interface
    PUBLIC
    FILE_SET HEADERS
    BASE_DIRS ${RFATTENUATOR_INTERFACE_PUBLIC_INCLUDE}
    FILES
        ${RFATTENUATOR_INTERFACE_PUBLIC_INCLUDE_PREFIX}/RfAttenuator.hxx
)

install(
    TARGETS ${PROJECT_NAME}-rfattenuator-interface
    EXPORT ${PROJECT_NAME}
    COMPONENT dev
    FILE_SET HEADERS
    PUBLIC_HEADER DESTINATION "${NETREMOTE_DIR_INSTALL_PUBLIC_HEADER_BASE}/{RFATTENUATOR-INTERFACE_PUBLIC_INCLUDE_SUFFIX}"
)
